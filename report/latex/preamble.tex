% ---------------------------- Colours --------------------------------

% ------------------------- Heading style -----------------------------
\titleformat{\section}
  {\color{sectionblue}\bfseries\large}{\thesection}{0.6em}{}
\titleformat{\subsection}
  {\bfseries\normalsize}{\thesubsection}{0.6em}{}

% "N marks" badge next to a task heading
\newcommand{\taskmarks}[1]{%
  \hspace{0.5em}\raisebox{0.12ex}{\colorbox{sectionblue}{\color{white}\scriptsize\,#1 marks\,}}}

% ------------------------- Page headers ------------------------------

\fancyhf{}

% ------------------------ Code listings -----------------------------
\lstdefinestyle{py}{
  language=Python,
  backgroundcolor=\color{codebg},
  basicstyle=\ttfamily\footnotesize,
  keywordstyle=\color{codekw}\bfseries,
  commentstyle=\color{codecomment}\itshape,
  stringstyle=\color{codestring},
  numbers=left,
  numberstyle=\tiny\color{codecomment},
  numbersep=8pt,
  frame=single,
  rulecolor=\color{gray!40},
  breaklines=true,
  showstringspaces=false,
  captionpos=b,
  xleftmargin=2.2em,
  framexleftmargin=2em,
  aboveskip=1em,
  belowskip=1em,
}
\lstset{style=py}

% ---------------------- Note / callout box --------------------------
